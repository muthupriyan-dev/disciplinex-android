# TruthLens AI — Docker build
# Render's native Python environment has a read-only filesystem for apt,
# so system dependencies (Tesseract OCR) require a Docker-based build.

FROM python:3.12-slim

# Install system dependency: Tesseract OCR (required by pytesseract for
# the Image (OCR) analysis feature), plus the Tamil language pack so
# Tamil-script news screenshots can be read (not just English).
RUN apt-get update && \
    apt-get install -y --no-install-recommends tesseract-ocr tesseract-ocr-tam && \
    rm -rf /var/lib/apt/lists/*

WORKDIR /app

COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY . .

# Render provides $PORT at runtime; default to 5000 for local docker runs
ENV PORT=5000
EXPOSE 5000

CMD gunicorn app:app --bind 0.0.0.0:$PORT
